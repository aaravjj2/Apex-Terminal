```python
import pytest
from pathlib import Path
from phase1.services.market_data.replay import (
    _canonical_key, has_replay, get_replay, save_replay, list_replays, clear_replays
)

def test_canonical_key_deterministic():
    """Canonical keys are stable for same params."""
    params1 = {"symbol": "AAPL", "start": "2024-01-01", "end": "2024-01-31"}
    params2 = {"end": "2024-01-31", "symbol": "AAPL", "start": "2024-01-01"}  # Different order
    
    key1 = _canonical_key("bars", params1)
    key2 = _canonical_key("bars", params2)
    
    assert key1 == key2, "Keys must be order-independent"
    assert key1.startswith("bars_"), "Keys must be prefixed with request type"

def test_replay_save_and_retrieve():
    """Replay artifacts can be saved and retrieved."""
    clear_replays()  # Clean slate
    
    params = {"symbol": "TEST", "start": "2024-01-01", "end": "2024-01-31"}
    data = {"bars": [{"open": 100, "close": 101}]}
    
    save_replay("bars", params, data)
    assert has_replay("bars", params)
    
    retrieved = get_replay("bars", params)
    assert retrieved is not None
    assert retrieved["data"] == data

def test_replay_blocks_provider_fetch(mocker):
    """When replay exists, provider fetch must NOT be invoked."""
    from phase1.services.market_data.providers import DemoProvider
    import asyncio
    
    # Save a replay
    params = {"symbol": "AAPL", "start": "2024-01-01", "end": "2024-01-31"}
    data = {"bars": [{"timestamp": "2024-01-15T00:00:00", "open": 150, "high": 151, "low": 149, "close": 150.5, "volume": 1000000}]}
    save_replay("bars", params, data)
    
    # Mock the fixture loading
    provider = DemoProvider(enable_replay_save=False)
    mock_load = mocker.patch.object(provider, '_load_fixture', return_value=[])
    
    # Request bars
    from phase1.services.market_data.providers.types import BarsRequest
    from datetime import datetime
    request = BarsRequest(symbol="AAPL", start=datetime(2024, 1, 1), end=datetime(2024, 1, 31), interval="1d")
    
    response = asyncio.run(provider.get_bars(request))
    
    # Fixture should NOT have been called
    mock_load.assert_not_called()
    assert response.cached is True
    assert len(response.bars) > 0
```

#### Task 6: Shared Schemas
**File:** `phase1/services/market_data/schemas.py` (create new)

```python
"""
Shared schemas for bars and quotes — enforced across all providers.
"""

from pydantic import BaseModel, Field, validator
from datetime import datetime
from typing import Optional

class BarSchema(BaseModel):
    """Canonical OHLCV bar schema."""
    timestamp: datetime = Field(..., description="Bar timestamp (UTC)")
    open: float = Field(..., gt=0, description="Open price (must be > 0)")
    high: float = Field(..., gt=0, description="High price")
    low: float = Field(..., gt=0, description="Low price")
    close: float = Field(..., gt=0, description="Close price")
    volume: int = Field(..., ge=0, description="Volume (must be >= 0)")
    
    @validator('high')
    def high_must_be_highest(cls, v, values):
        if 'low' in values and 'open' in values and 'close' in values:
            assert v >= values['low'], "High must be >= low"
            assert v >= values['open'], "High must be >= open"
            assert v >= values['close'], "High must be >= close"
        return v
    
    @validator('low')
    def low_must_be_lowest(cls, v, values):
        if 'open' in values and 'close' in values:
            assert v <= values['open'], "Low must be <= open"
            assert v <= values['close'], "Low must be <= close"
        return v

class QuoteSchema(BaseModel):
    """Canonical quote schema."""
    symbol: str = Field(..., min_length=1, max_length=10, description="Ticker symbol")
    timestamp: datetime = Field(..., description="Quote timestamp (UTC)")
    price: float = Field(..., gt=0, description="Last trade price")
    bid: Optional[float] = Field(None, gt=0, description="Bid price")
    ask: Optional[float] = Field(None, gt=0, description="Ask price")
    volume: Optional[int] = Field(None, ge=0, description="Daily volume")
    
    @validator('ask')
    def ask_must_be_gte_bid(cls, v, values):
        if v is not None and 'bid' in values and values['bid'] is not None:
            assert v >= values['bid'], "Ask must be >= bid"
        return v

class ErrorSchema(BaseModel):
    """Deterministic error response."""
    error_code: str = Field(..., description="Error code (e.g., INVALID_SYMBOL)")
    message: str = Field(..., description="Human-readable message")
    details: Optional[dict] = Field(None, description="Additional error details")
    timestamp: datetime = Field(default_factory=datetime.utcnow, description="Error timestamp")
    
    class Config:
        # Ensure deterministic JSON ordering
        json_encoders = {
            datetime: lambda v: v.isoformat()
        }
```

```tsx
import React from 'react';

export interface DataStateBannerProps {
  state: 'empty' | 'error' | 'loading';
  errorCode?: string;
  errorMessage?: string;
  testIdPrefix?: string;
}

export function DataStateBanner({ state, errorCode, errorMessage, testIdPrefix = 'data-state-banner' }: DataStateBannerProps) {
  if (state === 'empty') {
    return (
      <div 
        className="w-full bg-gray-100 border border-gray-300 rounded p-4 text-center"
        data-testid={`${testIdPrefix}-empty`}
      >
        <div className="text-gray-600">No data available</div>
        <div className="text-sm text-gray-500 mt-1">Adjust your request parameters and try again</div>
      </div>
    );
  }
  
  if (state === 'error') {
    return (
      <div 
        className="w-full bg-red-100 border border-red-300 rounded p-4"
        data-testid={`${testIdPrefix}-error`}
      >
        <div className="text-red-900 font-medium" data-testid={`${testIdPrefix}-error-code`}>
          {errorCode || 'ERROR'}
        </div>
        <div className="text-red-700 mt-1" data-testid={`${testIdPrefix}-error-message`}>
          {errorMessage || 'An error occurred while fetching data'}
        </div>
      </div>
    );
  }
  
  if (state === 'loading') {
    return (
      <div 
        className="w-full bg-blue-100 border border-blue-300 rounded p-4 text-center"
        data-testid={`${testIdPrefix}-loading`}
      >
        <div className="text-blue-900">Loading...</div>
      </div>
    );
  }
  
  return null;
}
```

```python
def test_quote_canonical_hash_stable():
```
